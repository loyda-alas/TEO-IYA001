JMP start

num:   DB 7		; Numero que vamos a probar
par:   DB "Es par"
       DB 0
impar: DB "Es impar"
       DB 0

start:
	MOV A, [num]

restar:
	CMP A, 0
	JZ es_par
	CMP A, 1
	JZ es_impar
	SUB A, 2
	JMP restar

es_par:
	MOV C, par
	JMP mostrar

es_impar:
	MOV C, impar

mostrar:
	MOV D, 232
	CALL print
	HLT

print:
	PUSH A
	PUSH B
	MOV B, 0
.loop:
	MOV A, [C]
	MOV [D], A
	INC C
	INC D  
	CMP B, [C]
	JNZ .loop

	POP B
	POP A
	RET
